Define $r:\mathbb{R}^5\to\mathbb{R}^5$ by $r(a,b,c,d,e)\coloneqq(e,d,c,b,a)$.
\begin{theorem}\label{thm:heldout}
For all real numbers $a,b,c,d,e$, $r(r(a,b,c,d,e))=(a,b,c,d,e)$.
\end{theorem}
\end{document}
