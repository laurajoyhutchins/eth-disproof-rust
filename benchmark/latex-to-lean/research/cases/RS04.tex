\section{spectral}
Define $e:\mathbb{R}^2\to\mathbb{R}$ by $e(x,y)\coloneqq\frac{x+y}{2}$.
Define $o:\mathbb{R}^2\to\mathbb{R}$ by $o(x,y)\coloneqq\frac{x-y}{2}$.
\begin{theorem}\label{thm:RS04}
For all real numbers $x,y$, $e(e(x,y),e(x,y))=e(x,y)$ and $o(o(x,y),-o(x,y))=o(x,y)$.
\end{theorem}
