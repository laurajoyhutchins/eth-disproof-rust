\section{groups}
Define $s:\mathbb{R}^2\to\mathbb{R}^2$ by $s(x,y)\coloneqq(y,x)$.
\begin{theorem}\label{thm:RG02}
For all real numbers $x,y$, $s(s(x,y))=(x,y)$.
\end{theorem}
