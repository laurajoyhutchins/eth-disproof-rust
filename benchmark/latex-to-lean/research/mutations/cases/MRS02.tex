\section{spectral}
Define $e:\mathbb{R}^2\to\mathbb{R}$ by $e(x,y)\coloneqq\frac{x+y}{2}$.
Define $o:\mathbb{R}^2\to\mathbb{R}$ by $o(x,y)\coloneqq\frac{x-y}{2}$.
\begin{theorem}\label{thm:RS02}
For all real numbers $x,y$, $e(x,y)+o(x,y)=x$ and $e(x,y)-o(x,y)=x$.
\end{theorem}
