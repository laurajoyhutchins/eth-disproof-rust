\section{groups}
Define $s:\mathbb{R}^4\to\mathbb{R}^4$ by $s(a,b,c,d)\coloneqq(b,a,c,d)$.
Define $t:\mathbb{R}^4\to\mathbb{R}^4$ by $t(a,b,c,d)\coloneqq(a,b,d,c)$.
\begin{theorem}\label{thm:RG04}
For every $v\in\mathbb{R}^4$, $(s\circ t)(v)=(t\circ s)(v)$.
\end{theorem}
