\section{groups}
Define $u:\mathbb{R}^2\to\mathbb{R}$ by $u(x,y)\coloneqq y$.
Define $v:\mathbb{R}^2\to\mathbb{R}$ by $v(x,y)\coloneqq x$.
\begin{theorem}\label{thm:RG01}
For all real numbers $x,y$, $u(u(x,y),v(x,y))=x$ and $v(u(x,y),v(x,y))=y$.
\end{theorem}
